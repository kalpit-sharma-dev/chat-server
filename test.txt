package main

import (
	"context"
	"fmt"
	"io"
	"log"
	"net/http"
	"sync"
	"time"
)

var (
	cancelFunc context.CancelFunc
	wg         sync.WaitGroup
	running    bool
	loopMutex  sync.Mutex
)

// Infinite loop function that spawns a goroutine to call an API
func startInfiniteLoop(ctx context.Context) {
	loopMutex.Lock()
	if running {
		log.Println("Loop already running, ignoring request.")
		loopMutex.Unlock()
		return
	}
	running = true
	loopMutex.Unlock()

	log.Println("Starting infinite loop...")

	for {
		select {
		case <-ctx.Done():
			log.Println("Stopping infinite loop and all goroutines...")
			running = false
			return
		default:
			wg.Add(1)
			go callAPI(ctx) // Goroutine making API call
			select {
			case <-ctx.Done():
				log.Println("Loop received stop signal, exiting.")
				running = false
				return
			case <-time.After(1 * time.Second): // Sleep before next iteration
			}
		}
	}
}

// Function to make an API call
func callAPI(ctx context.Context) {
	defer wg.Done()

	url := "https://jsonplaceholder.typicode.com/todos/1"

	// Check if context is done before making the request
	select {
	case <-ctx.Done():
		log.Println("API request skipped due to stop signal.")
		return
	default:
	}

	req, err := http.NewRequestWithContext(ctx, "GET", url, nil) // Attach context to the request
	if err != nil {
		log.Printf("Failed to create request: %v", err)
		return
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		log.Printf("API request failed: %v", err)
		return
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		log.Printf("Failed to read API response: %v", err)
		return
	}

	log.Printf("API Response: %s", body)
}

// API handler
func handler(w http.ResponseWriter, r *http.Request) {
	query := r.URL.Query().Get("call")

	if query == "true" {
		ctx, cancel := context.WithCancel(context.Background())
		cancelFunc = cancel
		go startInfiniteLoop(ctx)
		w.WriteHeader(http.StatusOK)
		fmt.Fprintln(w, "Started infinite loop with API calls.")
	} else if query == "false" {
		if cancelFunc != nil {
			cancelFunc() // Stop all goroutines immediately
		}
		wg.Wait() // Wait for all goroutines to complete
		w.WriteHeader(http.StatusOK)
		fmt.Fprintln(w, "All goroutines stopped.")
	} else {
		http.Error(w, "Invalid query parameter", http.StatusBadRequest)
	}
}

// Main function
func main() {
	http.HandleFunc("/api", handler)

	log.Println("Server started on :8080")
	if err := http.ListenAndServe(":8080", nil); err != nil {
		log.Fatalf("Server error: %v", err)
	}
}
