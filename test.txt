package main

import (
	"fmt"
	"io"
	"log"
	"net/http"
	"sync"
	"time"
)

var (
	stopLoop    = make(chan struct{}) // Channel to signal stopping the infinite loop
	wg          sync.WaitGroup        // WaitGroup to track goroutines
	running     bool                  // Flag to track infinite loop status
	routineLock sync.Mutex            // Mutex to synchronize goroutine count
)

// Infinite loop function that spawns a goroutine to call an API
func startInfiniteLoop() {
	log.Println("Starting infinite loop...")
	running = true

	for running {
		select {
		case <-stopLoop:
			log.Println("Stopping infinite loop...")
			running = false
			return
		default:
			wg.Add(1)
			go callAPI() // Goroutine making API call
			time.Sleep(1 * time.Second) // Wait before next iteration
		}
	}
}

// Function to make an API call
func callAPI() {
	defer wg.Done()

	url := "https://jsonplaceholder.typicode.com/todos/1" // Example API
	resp, err := http.Get(url)
	if err != nil {
		log.Printf("API request failed: %v", err)
		return
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		log.Printf("Failed to read API response: %v", err)
		return
	}

	log.Printf("API Response: %s", body)
}

// API handler
func handler(w http.ResponseWriter, r *http.Request) {
	query := r.URL.Query().Get("call")

	if query == "true" {
		go startInfiniteLoop()
		w.WriteHeader(http.StatusOK)
		fmt.Fprintln(w, "Started infinite loop with API calls.")
	} else if query == "false" {
		close(stopLoop) // Stop infinite loop
		wg.Wait()       // Wait for all goroutines to complete
		w.WriteHeader(http.StatusOK)
		fmt.Fprintln(w, "All goroutines stopped.")
	} else {
		http.Error(w, "Invalid query parameter", http.StatusBadRequest)
	}
}

// Main function
func main() {
	http.HandleFunc("/api", handler)

	log.Println("Server started on :8080")
	if err := http.ListenAndServe(":8080", nil); err != nil {
		log.Fatalf("Server error: %v", err)
	}
}
