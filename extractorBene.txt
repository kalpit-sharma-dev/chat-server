📄 parser.go – Transaction Row Parser

package main

import (
	"regexp"
	"strconv"
	"strings"
)

var dateRegex = regexp.MustCompile(`^\d{2}/\d{2}/\d{2}`)

func ParseTransactions(text string) []Transaction {
	lines := strings.Split(text, "\n")
	var txns []Transaction

	for _, line := range lines {
		line = strings.TrimSpace(line)
		if !dateRegex.MatchString(line) {
			continue
		}

		fields := strings.Fields(line)
		if len(fields) < 5 {
			continue
		}

		date := fields[0]
		narration := strings.Join(fields[1:len(fields)-2], " ")

		amount, txnType := extractAmountAndType(line)

		txns = append(txns, Transaction{
			Date:      date,
			Narration: narration,
			Amount:    amount,
			Type:      txnType,
		})
	}

	return txns
}

func extractAmountAndType(line string) (float64, string) {
	amountRegex := regexp.MustCompile(`([\d,]+\.\d{2})`)
	matches := amountRegex.FindAllString(line, -1)

	if len(matches) == 0 {
		return 0, "UNKNOWN"
	}

	amountStr := strings.ReplaceAll(matches[len(matches)-1], ",", "")
	amount, _ := strconv.ParseFloat(amountStr, 64)

	// crude but works well for Indian statements
	if strings.Contains(line, "CR") || strings.Contains(line, "Deposit") {
		return amount, "CREDIT"
	}

	return amount, "DEBIT"
}

📄 extractor.go – Beneficiary Extraction Logic

package main

import (
	"regexp"
	"strings"
)

var (
	upiRegex  = regexp.MustCompile(`UPI-([A-Z .&]+)`)
	impsRegex = regexp.MustCompile(`IMPS-\d+-MR ([A-Z ]+)`)
	posRegex  = regexp.MustCompile(`POS ([A-Z0-9 .]+)`)
)

func NormalizeNarration(s string) string {
	s = strings.ToUpper(s)

	patterns := []string{
		`\d{10,}`,
		`@[A-Z0-9]+`,
		`-UPI|-IMPS|-NEFT|-POS`,
		`HDFC0MERUPI|YESB0PTMUPI`,
	}

	for _, p := range patterns {
		re := regexp.MustCompile(p)
		s = re.ReplaceAllString(s, "")
	}

	return strings.Join(strings.Fields(s), " ")
}

func ExtractBeneficiary(narration string) string {
	n := NormalizeNarration(narration)

	if m := upiRegex.FindStringSubmatch(n); len(m) > 1 {
		return strings.TrimSpace(m[1])
	}

	if m := impsRegex.FindStringSubmatch(n); len(m) > 1 {
		return strings.TrimSpace(m[1])
	}

	if m := posRegex.FindStringSubmatch(n); len(m) > 1 {
		return strings.TrimSpace(m[1])
	}

	if strings.Contains(n, "SALARY") {
		return "EMPLOYER"
	}

	return "UNKNOWN"
}

📄 classifier.go – Entity Classification

package main

import "strings"

func ClassifyEntity(name string) string {
	name = strings.ToUpper(name)

	merchantKeywords := []string{
		"STORE", "MART", "DAIRY", "SWEET", "FUEL",
		"GOOGLE", "AMAZON", "ZERODHA", "SPOTIFY",
		"AIRTEL", "PAYTM", "ZEPTO", "BLINKIT",
	}

	for _, k := range merchantKeywords {
		if strings.Contains(name, k) {
			return "MERCHANT"
		}
	}

	if strings.HasPrefix(name, "MR ") ||
		strings.Contains(name, " KUMAR") ||
		strings.Contains(name, " SINGH") {
		return "PERSON"
	}

	if name == "EMPLOYER" {
		return "BANK"
	}

	return "UNKNOWN"
}

📄 main.go – End-to-End Runner

package main

import (
	"fmt"
	"os"
)

func main() {

	// 🔹 Replace this with PDF → text output
	data, err := os.ReadFile("statement.txt")
	if err != nil {
		panic(err)
	}

	text := string(data)

	transactions := ParseTransactions(text)

	for i := range transactions {
		name := ExtractBeneficiary(transactions[i].Narration)
		entity := ClassifyEntity(name)

		transactions[i].Beneficiary = name
		transactions[i].EntityType = entity
	}

	// Print results
	for _, t := range transactions {
		fmt.Printf(
			"%s | %-6s | ₹%-8.2f | %-20s | %s\n",
			t.Date,
			t.Type,
			t.Amount,
			t.Beneficiary,
			t.EntityType,
		)
	}
}
